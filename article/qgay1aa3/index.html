127字小于1分钟

2024-06-06

VuePress开启PWA

创建manifest

.vuepress/public文件下创建manifest.json

{
    "lang": "zh-cn",
    "name": "前端笔记",
    "short_name": "前端笔记",
    "description": "前端学习笔记",
    "start_url": "/",
    "background_color": "#2f3d58",
    "theme_color": "#2f3d58",
    "orientation": "any",
    "display": "standalone",
    "icons": [
        {
            "src": "/logo.png",
            "sizes": "144x144"
        }
    ]
}

加载manifest

.vuepress/config.js添加一下代码


export default defineUserConfig({
    ...
    ...
     head: [
        // 引入manifest 开启PWA
        ['link', {
            rel: 'manifest',
            href: '/manifest.json'
        }]
    ],
})

开启Service Worker

.vuepress/config.js添加一下代码


export default defineUserConfig({
    ...
    ...
    serviceWorker:true
})
